import React, { useState } from 'react';
import './style.css';
import {
  NAV_ITEMS, MOCK_USERS, MOCK_CLIENTS, MOCK_CARS, MOCK_APPLICATION, MOCK_SERVICE_STATIONS, MOCK_STATUSES, MOCK_SERVICES,
  MOCK_STAFF, MOCK_VENDORS, MOCK_WORK_ORDERS, MOCK_SPARE_PARTS_CONTRACTS, MOCK_WORK_ORDER_SERVICES, MOCK_WORK_ORDER_SPARE_PARTS,
  MOCK_SPARE_PARTS_CONTRACT_ITEMS
} from './data/mockData';
import WorkOrderModal from './components/modals/WorkOrderModal';
import LoginForm from './components/layout/LoginForm';

function App() {

  // Текущий пользователь (изначально пустой)
  const [currentUser, setCurrentUser] = useState(null);

  const [activeTab, setActiveTab] = useState('dashboard');
  const [isModalOpen, setIsModalOpen] = useState(false);
  // Состояние для основных атрибутов ЗН

  const getPageTitle = (tab) => {
    switch (tab) {
      case 'dashboard':
        return 'Рабочий стол';
      case 'work-orders':
        return 'Список ЗН';
      case 'applications':
        return 'Заявки';
      case 'clients':
        return 'База клиентов';
      case 'cars':
        return 'Автопарк';
      case 'staff':
        return 'Персонал';
      case 'service-stations':
        return 'Филиалы';
      case 'service-catalog':
        return 'Каталог услуг';
      case 'vendors':
        return 'Поставщики';
      case 'contracts':
        return 'Закупки';
      default:
        return 'Система СТО';
    }
  };



  // Управление вкладками
  const handleTabClick = (tab) => setActiveTab(tab);

  const handleLoginSuccess = (user) => {
    setCurrentUser(user);
  };

  return (
    <div className="app-container">
      {!currentUser ? (
        <LoginForm onLoginSuccess={handleLoginSuccess} />
      ) : (
        /* ОСНОВНОЙ ИНТЕРФЕЙС (Твой текущий код) */
        <>
          <nav className="sidebar">
            <div className="sidebar-header" style={{ padding: '20px 0' }}>
              <span className="logo-text">СТО Система</span>
              <div style={{ color: '#94a3b8', fontSize: '11px', marginTop: '10px' }}>
                Вы вошли как: <br /><strong>{currentUser?.name}</strong>
              </div>
              <button
                onClick={() => {setCurrentUser(null);}}
                style={{ marginTop: '10px', fontSize: '10px', padding: '4px' }}
              >Выйти</button>
            </div>

            <ul style={{ listStyle: 'none', padding: 0 }}>
              {NAV_ITEMS
                .filter(item => item.roles.includes(currentUser.role))
                .map(item => (
                  <li
                    key={item.id}
                    className={activeTab === item.id ? 'active' : ''}
                    onClick={() => setActiveTab(item.id)}
                    style={{ padding: '12px 20px', cursor: 'pointer' }}
                  >
                    {item.label}
                  </li>
                ))
              }
            </ul>
          </nav>

          <main className="content">
            <header className="content-header">
              <h1>{NAV_ITEMS.find(i => i.id === activeTab)?.label || 'Система'}</h1>
              <div className="header-actions">

                {activeTab === 'work-orders' && currentUser.role === 'advisor' && (
                  <button
                    className="btn-primary"
                    onClick={() => {
                      setIsModalOpen(true);
                    }}
                  >
                    + Создать ЗН
                  </button>
                )}
              </div>
            </header>

            <section id="view-container">
              {activeTab === 'dashboard' && (
                <div className="dashboard-grid">
                  <div className="dashboard-card">
                    <h3>Профиль: {currentUser.name}</h3>
                    <p>Статус: Постоянный клиент</p>
                  </div>

                  {currentUser.role === 'client' && (
                    <>
                      <div className="dashboard-card highlight">
                        <h3>В работе</h3>
                        {/* Считаем ЗН со статусом "В работе" для этого клиента */}
                        <div className="big-number">2 автомобиля</div>
                      </div>
                      <div className="dashboard-card">
                        <h3>Ваши бонусы</h3>
                        <p>500 баллов</p>
                      </div>
                    </>
                  )}
                </div>
              )}

              {activeTab === 'work-orders' && (
                <table className="data-table">
                  <thead>
                    <tr>
                      <th>ID</th><th>ID заявки</th><th>Мастер-приёмщик</th><th>Филиал</th><th>Статус</th><th>Итоговая цена комплектующих</th>
                      <th>Итоговая цена услуг</th><th>К оплате</th><th>Дата закрытия ЗН</th>
                    </tr>
                  </thead>
                  <tbody>
                    {MOCK_WORK_ORDERS.filter(order => {
                      // Если залогинен клиент — показываем только его заказы
                      if (currentUser.role === 'client') {
                        const app = MOCK_APPLICATION.find(a => a.id === order.application_id);
                        const car = MOCK_CARS.find(c => c.id === app?.car_id);
                        return car?.owner_id === currentUser.id;
                      }
                      return true; // Остальные роли видят всё
                    }).map(order => {
                      // 1. Считаем запчасти
                      const partsTotal = MOCK_WORK_ORDER_SPARE_PARTS
                        .filter(p => p.work_order_id === order.id)
                        .reduce((sum, p) => sum + (p.applied_price * p.count), 0);

                      // 2. Считаем услуги
                      const servicesTotal = MOCK_WORK_ORDER_SERVICES
                        .filter(s => s.work_order_id === order.id)
                        .reduce((sum, s) => sum + (s.applied_price * s.count * s.coeff), 0);

                      // 3. Находим мастера и филиал для отображения имен, а не ID
                      const advisor = MOCK_STAFF.find(s => s.id === order.service_advisor_id);
                      const station = MOCK_SERVICE_STATIONS.find(st => st.id === order.service_station_id);
                      const status = MOCK_STATUSES.find(s => s.id === order.status_id);

                      return (
                        <tr key={order.id}>
                          <td>ЗН-{order.id}</td>
                          <td>№{order.application_id}</td>
                          <td>{advisor ? `${advisor.surname} ${advisor.name[0]}.` : '—'}</td>
                          <td>{station ? station.city : '—'}</td>
                          <td>
                            <span className={`status-badge status-${order.status_id}`}>
                              {status?.name}
                            </span>
                          </td>
                          <td>{partsTotal.toLocaleString()} руб.</td>
                          <td>{servicesTotal.toLocaleString()} руб.</td>
                          <td><strong>{(partsTotal + servicesTotal).toLocaleString()} руб.</strong></td>
                          <td>{order.close_date || '—'}</td>
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              )}

              {activeTab === 'clients' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Фамилия</th><th>Имя</th><th>Отчество</th><th>Телефон</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_CLIENTS.map(c => (
                      <tr key={c.id}>
                        <td>{c.surname}</td>
                        <td>{c.name}</td>
                        <td>{c.patronymic}</td>
                        <td>{c.phone}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}

              {activeTab === 'cars' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Владелец</th><th>Марка</th><th>Модель</th><th>Год выпуска</th><th>VIN</th><th>Гос. номер</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_CARS
                      .filter(car => currentUser.role === 'client' ? car.owner_id === currentUser.id : true)
                      .map(car => (
                        <tr key={car.id}>
                          <td>{car.owner_id}</td>
                          <td>{car.brand}</td>
                          <td>{car.model}</td>
                          <td>{car.production_year}</td>
                          <td>{car.vin}</td>
                          <td>{car.license_plate}</td>
                        </tr>
                      ))
                    }
                  </tbody>
                </table>
              )}

              {activeTab === 'applications' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Машина</th><th>Администратор, оформивший заявку</th><th>Описание проблем</th><th>Дата регистрации</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_APPLICATION.map(c => (
                      <tr key={c.id}>
                        <td>{c.car_id}</td>
                        <td>{c.administrator_id}</td>
                        <td>{c.troubles_description}</td>
                        <td>{c.registration_date}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
              {activeTab === 'staff' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Филиал</th><th>Фамилия</th><th>Имя</th><th>Отчество</th><th>Должность</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_STAFF.map(c => (
                      <tr key={c.id}>
                        <td>{c.service_station_id}</td>
                        <td>{c.surname}</td>
                        <td>{c.name}</td>
                        <td>{c.patronymic}</td>
                        <td>{c.role}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
              {activeTab === 'service-stations' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Регион</th><th>Город</th><th>Улица</th><th>дом</th><th>Телефон</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_SERVICE_STATIONS.map(c => (
                      <tr key={c.id}>
                        <td>{c.region}</td>
                        <td>{c.city}</td>
                        <td>{c.street}</td>
                        <td>{c.house}</td>
                        <td>{c.phone_number}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
              {activeTab === 'service-catalog' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Название</th><th>Цена</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_SERVICES.map(c => (
                      <tr key={c.id}>
                        <td>{c.name}</td>
                        <td>{c.price}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
              {activeTab === 'vendors' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Регион</th><th>Город</th><th>Улица</th><th>Дом</th><th>Номер офиса/квартиры</th><th>Почтовый индекс</th>
                      <th>ИНН</th><th>Телефон</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_VENDORS.map(c => (
                      <tr key={c.id}>
                        <td>{c.region}</td>
                        <td>{c.city}</td>
                        <td>{c.street}</td>
                        <td>{c.house}</td>
                        <td>{c.office_flat}</td>
                        <td>{c.postcode}</td>
                        <td>{c.TIN}</td>
                        <td>{c.phone_number}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
              {activeTab === 'contracts' && (
                <table className="data-table">
                  <thead>
                    <tr><th>Поставщик</th><th>Дата</th><th>Итоговая цена</th></tr>
                  </thead>
                  <tbody>
                    {MOCK_SPARE_PARTS_CONTRACTS.map(c => (
                      <tr key={c.id}>
                        <td>{c.vendor_id}</td>
                        <td>{c.date}</td>
                        <td>{c.total_price}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
            </section>
          </main>

          <WorkOrderModal
            isOpen={isModalOpen}
            onClose={() => setIsModalOpen(false)}
            applications={MOCK_APPLICATION}
            staff={MOCK_STAFF}
            stations={MOCK_SERVICE_STATIONS}
            statuses={MOCK_STATUSES}
            services={MOCK_SERVICES}
            clients={MOCK_CLIENTS}
            cars={MOCK_CARS}
          />
        </>
      )}
    </div>
  );
}

export default App;
