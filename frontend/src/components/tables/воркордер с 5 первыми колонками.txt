import React from 'react';
import {
  MOCK_WORK_ORDERS,
  MOCK_STAFF,
  MOCK_SERVICE_STATIONS,
  MOCK_STATUSES,
  MOCK_APPLICATION // Добавили импорт заявок для поиска филиала
} from '../../data/mockData';

const WorkOrderTable = () => {
  return (
    <table className="data-table">
      <thead>
        <tr>
          <th>ID</th>
          <th>ID заявки</th>
          <th>Мастер-приёмщик</th>
          <th>Филиал</th>
          <th>Статус</th>
          <th>Итоговая цена комплектующих</th>
          <th>Итоговая цена услуг</th>
          <th>К оплате</th>
          <th>Дата закрытия ЗН</th>
        </tr>
      </thead>
      <tbody>
        {MOCK_WORK_ORDERS.map(order => {
          // 1. Сначала находим заявку, так как филиал и мастер могут быть в ней
          const application = MOCK_APPLICATION.find(a => a.id == order.application_id);

          // 2. Ищем Мастера: пробуем advisor_id из заказа или administrator_id из заявки
          const staffId = order.advisor_id || order.master_id || application?.administrator_id;
          const staff = MOCK_STAFF.find(s => s.id == staffId);

          // 3. Ищем Филиал: пробуем station_id из заказа или service_station_id из заявки/мастера
          const stationId = order.station_id || application?.service_station_id || staff?.service_station_id;
          const station = MOCK_SERVICE_STATIONS.find(s => s.id == stationId);

          // 4. Статус (уже работал, оставляем)
          const status = MOCK_STATUSES.find(s => s.id == order.status_id);

          // 5. Цены: если поля пустые, ставим 0, чтобы не было пустоты в таблице
          const partsPrice = order.total_spare_parts_price ?? order.spare_parts_cost ?? 0;
          const servicesPrice = order.total_services_price ?? order.services_cost ?? 0;
          const totalPrice = order.total_price ?? (Number(partsPrice) + Number(servicesPrice));

          // 6. Дата: пробуем разные варианты ключа даты
          const closingDate = order.date || order.closing_date || order.end_date || '—';

          return (
            <tr key={order.id}>
              <td>{order.id}</td>
              <td>{order.application_id}</td>
              <td>{staff ? `${staff.surname} ${staff.name}` : 'Н/Д'}</td>
              <td>{station ? (
                `${station.city || ''}, ${station.street || ''}, д. ${station.house || ''}`
              ) : 'Н/Д'}</td>
              <td>
                <span className={`status-badge status-${order.status_id}`}>
                  {status ? status.name : 'Н/Д'}
                </span>
              </td>
              <td>{partsPrice}</td>
              <td>{servicesPrice}</td>
              <td>{totalPrice}</td>
              <td>{closingDate}</td>
            </tr>
          );
        })}
      </tbody>
    </table>
  );
};

export default WorkOrderTable;
