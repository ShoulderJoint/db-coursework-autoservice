autoservice_kondratev-название

CREATE TABLE system_roles (
    id SERIAL PRIMARY KEY,
    system_name VARCHAR(50) not null,
    display_name VARCHAR(100) NOT NULL,
    unique(system_name)
);

CREATE TABLE order_statuses (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description VARCHAR(255) NOT NULL
);

CREATE TABLE services (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    price DECIMAL(12, 2) NOT NULL, check (price>=0)
);

CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

-- Таблица филиалов/станций
CREATE TABLE stations (
    id SERIAL PRIMARY KEY,
    region VARCHAR(100) NOT NULL,
    city VARCHAR(100) NOT NULL,
    street VARCHAR(100) NOT NULL,
    house VARCHAR(20) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

-- Таблица поставщиков запчастей
CREATE TABLE vendors (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    region VARCHAR(100) NOT NULL,
    city VARCHAR(100) NOT NULL,
    street VARCHAR(100) NOT NULL,
    house VARCHAR(20) NOT NULL,
    flat VARCHAR(20),
    postcode CHAR(6),
    inn VARCHAR(12) NOT NULL,
    phone VARCHAR(20)
);

-- Таблица клиентов
CREATE TABLE clients (
    id SERIAL PRIMARY KEY,
    surname VARCHAR(100) NOT NULL,
    name VARCHAR(100) NOT NULL,
    patronymic VARCHAR(100),
    phone VARCHAR(20) NOT NULL,
    login VARCHAR(50) NOT NULL,
    password_hash BYTEA NOT NULL,
    system_role_id INT REFERENCES system_roles(id),
    unique(login)
);

-- Таблица сотрудников
CREATE TABLE staff (
    id SERIAL PRIMARY KEY,
    station_id INT NOT NULL REFERENCES stations(id),
    surname VARCHAR(100) NOT NULL,
    name VARCHAR(100) NOT NULL,
    patronymic VARCHAR(100),
    role_id INT NOT NULL REFERENCES roles(id),
    login VARCHAR(50),
    password_hash BYTEA,
    system_role_id INT REFERENCES system_roles(id),
    unique(login)
);

-- Таблица автомобилей
CREATE TABLE cars (
    id SERIAL PRIMARY KEY,
    client_id INT NOT NULL REFERENCES clients(id),
    brand VARCHAR(100) NOT NULL,
    model VARCHAR(100) NOT NULL,
    production_year smallint NOT NULL, CHECK (production_year >= 1886),
    vin VARCHAR(17) NOT NULL,
    reg_number VARCHAR(20) NOT NULL,
    unique(vin),
    unique(reg_number)
);

-- Таблица заявок на ремонт
CREATE TABLE applications (
    id SERIAL PRIMARY KEY,
    car_id INT NOT NULL REFERENCES cars(id),
    staff_id INT NOT NULL REFERENCES staff(id),
    description VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP,
    check (updated_at>created_at)
);

-- Таблица заказов (основная информация)
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    application_id INT NOT NULL REFERENCES applications(id),
    staff_id INT NOT NULL REFERENCES staff(id),
    status_id INT NOT NULL REFERENCES order_statuses(id),
    cost_parts DECIMAL(12, 2) NOT NULL DEFAULT 0,
    cost_services DECIMAL(12, 2) NOT NULL DEFAULT 0,
    cost DECIMAL(12, 2) NOT NULL DEFAULT 0,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    closed_at TIMESTAMP,
    check (closed_at>created_at)
);

-- Таблица услуг в заказе
CREATE TABLE order_service (
    id SERIAL PRIMARY KEY,
    service_id INT NOT NULL REFERENCES services(id),
    work_order_id INT NOT NULL REFERENCES orders(id),
    count INT NOT NULL DEFAULT 1 check(count>0),
    coefficient DECIMAL(5, 2) NOT NULL DEFAULT 1.00 check(coefficient>0),
    cost DECIMAL(12, 2) NOT NULL
);

-- Таблица контрактов с поставщиками
CREATE TABLE parts_contracts (
    id SERIAL PRIMARY KEY,
    vendor_id INT NOT NULL REFERENCES vendors(id),
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP,
    document BYTEA,
    check(updated_at>created_at)
);

-- Таблица позиций (запчастей) в контракте
CREATE TABLE parts_contract_items (
    id SERIAL PRIMARY KEY,
    contract_id INT NOT NULL REFERENCES parts_contracts(id),
    name VARCHAR(255) NOT NULL,
    count INT NOT NULL check(count>0)
);

-- Таблица запчастей, использованных в заказе
CREATE TABLE order_parts (
    id SERIAL PRIMARY KEY,
    work_order_id INT NOT NULL REFERENCES orders(id),
    spare_part_id INT NOT NULL REFERENCES parts_contract_items(id),
    count INT NOT NULL check(count>0),
    cost DECIMAL(12, 2) NOT NULL check(cost>=0)
);
